1//! Custom application error types and Axum HTTP response implementations.
2//!
3//! Maps internal application failures, database errors, authentication issues, and serialization
4//! errors into structured HTTP status codes and JSON error responses.
5
6use std::io;
7
8use ::hams::error::HamsError;
9use axum::extract::rejection::JsonRejection;
10use thiserror::Error;
11use tracing_subscriber::filter::FromEnvError;
12
13/// Root error type for backend operations and API responses.
14#[derive(Error, Debug)]
15pub enum AppError {
16    /// General internal server error message.
17    #[error("General error `{0}`")]
18    Message(String),
19
20    /// Client request invalid error (400 Bad Request).
21    #[error("Bad Request `{0}`")]
22    BadRequest(String),
23
24    /// Operation or service cancelled.
25    #[error("Service Cancelled")]
26    Cancelled,
27
28    /// Missing or invalid authentication credentials (401 Unauthorized).
29    #[error("Unauthorized `{0}`")]
30    Unauthorized(String),
31
32    /// Insufficient role or access permissions (403 Forbidden).
33    #[error("Forbidden `{0}`")]
34    Forbidden(String),
35
36    /// Error originating from Health Monitoring System (HaMS).
37    #[error("HaMs error `{0}`")]
38    HamsError(#[from] HamsError),
39
40    /// Prometheus recorder or exporter error.
41    #[error("Prometheus error `{0}`")]
42    PrometheusError(#[from] prometheus::Error),
43
44    /// JSON serialization or deserialization error.
45    #[error("Serdes error `{0}`")]
46    Serde(#[from] serde_json::Error),
47
48    /// I/O operation error.
49    #[error("data store disconnected")]
50    Io(#[from] io::Error),
51
52    /// Axum JSON extractor rejection error.
53    #[error("Json Rejection `{0}`")]
54    JsonRejection(#[from] JsonRejection),
55
56    /// Service shutdown health check failure.
57    #[error("Shutdown error")]
58    ShutdownCheck,
59
60    /// Service pre-flight startup check failure.
61    #[error("PreFlight error")]
62    PreflightCheck,
63
64    /// Configuration parsing error from Figment.
65    #[error("Figment error `{0}`")]
66    FigmentError(#[from] Box<figment::error::Error>),
67
68    /// Environment log filter parsing error.
69    #[error("EnvFilter error `{0}`")]
70    EnvFilterError(#[from] FromEnvError),
71
72    /// Requested resource was not found (404 Not Found).
73    #[error("Not Found: `{0}`")]
74    NotFound(String),
75
76    /// Database schema version mismatch error.
77    #[error("Schema Mismatch: expected {expected}, actual {actual}")]
78    SchemaMismatch {
79        /// Expected schema version.
80        expected: i32,
81        /// Actual detected schema version.
82        actual: i32,
83    },
84
85    /// Invalid HTTP header value error.
86    #[error("Invalid header value")]
87    InvalidHeaderValue(#[from] reqwest::header::InvalidHeaderValue),
88
89    /// PostgreSQL database query or connection pool error.
90    #[error("Database error")]
91    DatabaseError(#[from] sqlx::Error),
92}
93
94impl axum::response::IntoResponse for AppError {
95    // PRD Reference: 0001
96    fn into_response(self) -> axum::response::Response {
97        #[derive(serde::Serialize)]
98        struct ErrorResponse {
99            message: String,
100        }
101
102        let (status, message) = match self {
103            AppError::Message(msg) => (reqwest::StatusCode::INTERNAL_SERVER_ERROR, msg.to_string()),
104            AppError::BadRequest(msg) => (reqwest::StatusCode::BAD_REQUEST, msg.to_string()),
105            AppError::NotFound(msg) => (reqwest::StatusCode::NOT_FOUND, msg.to_string()),
106            AppError::SchemaMismatch { .. } => (
107                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
108                "Schema Mismatch".to_string(),
109            ),
110            AppError::Cancelled => (
111                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
112                "Cancelled".to_string(),
113            ),
114            AppError::Unauthorized(msg) => (reqwest::StatusCode::UNAUTHORIZED, msg.to_string()),
115            AppError::Forbidden(msg) => (reqwest::StatusCode::FORBIDDEN, msg.to_string()),
116            AppError::HamsError(_error) => (
117                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
118                "Hams Error".to_string(),
119            ),
120            AppError::Serde(_error) => {
121                (reqwest::StatusCode::BAD_REQUEST, "Serde Error".to_string())
122            }
123            AppError::Io(_error) => (
124                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
125                "IO Error".to_string(),
126            ),
127            AppError::ShutdownCheck => (
128                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
129                "Shutdown Check Failed".to_string(),
130            ),
131            AppError::PreflightCheck => (
132                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
133                "Preflight Check Failed".to_string(),
134            ),
135            AppError::FigmentError(_error) => (
136                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
137                "Config Error".to_string(),
138            ),
139            AppError::JsonRejection(rejection) => (rejection.status(), rejection.body_text()),
140            AppError::PrometheusError(_error) => (
141                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
142                "Prometheus Error".to_string(),
143            ),
144            AppError::EnvFilterError(_error) => (
145                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
146                "EnvFilter Error".to_string(),
147            ),
148            AppError::InvalidHeaderValue(_error) => (
149                reqwest::StatusCode::INTERNAL_SERVER_ERROR,
150                "Invalid Header Value".to_string(),
151            ),
152            AppError::DatabaseError(error) => {
153                tracing::error!("Database error: {}", error);
154                (
155                    reqwest::StatusCode::INTERNAL_SERVER_ERROR,
156                    "Database Error".to_string(),
157                )
158            }
159        };
160
161        (status, axum::Json(ErrorResponse { message })).into_response()
162    }
163}
