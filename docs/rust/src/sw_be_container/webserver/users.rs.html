9pub async fn list_users(State(state): State<AppState>) -> Result<Json<Vec<User>>, AppError> {
10    let env = state.config.debugging.environment.as_str();
11    let dev_auth = state.config.debugging.enable_dev_auth;
12    if env != "development" && env != "testing" && !dev_auth {
13        return Err(AppError::Forbidden(
14            "User directory listing is disabled in this environment".to_string(),
15        ));
16    }
17
18    let users =
19        sqlx::query_as::<_, User>("SELECT u.id, u.name, u.email, u.role, u.phone, u.description, u.is_suspended, u.client_log_level, u.keycloak_sub, ARRAY_AGG(m.name) FILTER (WHERE m.name IS NOT NULL) AS modules FROM users u LEFT JOIN user_modules um ON u.id = um.user_id LEFT JOIN modules m ON um.module_id = m.id GROUP BY u.id")
20            .fetch_all(&state.db_pool)
21            .await;
22    Ok(Json(users?))
23}

26pub async fn create_user(
27    State(state): State<AppState>,
28    caller: crate::webserver::auth::OptionalJwtUser,
29    Json(mut user): Json<User>,
30) -> Result<Json<User>, AppError> {
31    if user.keycloak_sub.is_none() {
32        if let Some(c) = caller.0 {
33            if !c.sub.is_empty() && c.sub.parse::<i64>().is_err() {
34                user.keycloak_sub = Some(c.sub);
35            }
36        }
37    }
38    let mut tx = state.db_pool.begin().await?;
39    let log_level = if user.client_log_level.is_empty() {
40        "INFO"
41    } else {
42        &user.client_log_level
43    };
44
45    let effective_sub = user.keycloak_sub.or_else(|| {
46        if user.id > 0 {
47            Some(user.id.to_string())
48        } else {
49            None
50        }
51    });
52
53    let new_user: User = loop {
54        if user.id > 0 {
55            let inserted = sqlx::query_as::<_, User>(
56                "INSERT INTO users (id, name, email, role, phone, description, is_suspended, client_log_level, keycloak_sub) VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9) RETURNING id, name, email, role, phone, description, is_suspended, client_log_level, keycloak_sub, NULL AS modules",
57            )
58            .bind(user.id)
59            .bind(&user.name)
60            .bind(&user.email)
61            .bind(&user.role)
62            .bind(&user.phone)
63            .bind(&user.description)
64            .bind(user.is_suspended)
65            .bind(log_level)
66            .bind(&effective_sub)
67            .fetch_one(&mut *tx)
68            .await?;
69
70            break inserted;
71        } else {
72            let res = sqlx::query_as::<_, User>(
73                "INSERT INTO users (name, email, role, phone, description, is_suspended, client_log_level, keycloak_sub) VALUES ($1, $2, $3, $4, $5, $6, $7, $8) RETURNING id, name, email, role, phone, description, is_suspended, client_log_level, keycloak_sub, NULL AS modules",
74            )
75            .bind(&user.name)
76            .bind(&user.email)
77            .bind(&user.role)
78            .bind(&user.phone)
79            .bind(&user.description)
80            .bind(user.is_suspended)
81            .bind(log_level)
82            .bind(&effective_sub)
83            .fetch_one(&mut *tx)
84            .await;
85
86            match res {
87                Ok(inserted) => break inserted,
88                Err(sqlx::Error::Database(db_err)) if db_err.constraint() == Some("users_pkey") => {
89                    // Sequence counter generated an ID that collides with an explicit ID; increment sequence and retry within the transaction
90                    sqlx::query("SELECT nextval(pg_get_serial_sequence('users', 'id'))")
91                        .execute(&mut *tx)
92                        .await?;
93                    continue;
94                }
95                Err(e) => return Err(AppError::from(e)),
96            }
97        }
98    };
99
100    if let Some(modules) = &user.modules {
101        if !modules.is_empty() {
102            sqlx::query("INSERT INTO user_modules (user_id, module_id) SELECT $1, id FROM modules WHERE name = ANY($2)")
103                .bind(new_user.id)
104                .bind(modules)
105                .execute(&mut *tx)
106                .await?;
107        }
108    }
109
110    let final_user = sqlx::query_as::<_, User>(
111        "SELECT u.id, u.name, u.email, u.role, u.phone, u.description, u.is_suspended, u.client_log_level, u.keycloak_sub, ARRAY_AGG(m.name) FILTER (WHERE m.name IS NOT NULL) AS modules FROM users u LEFT JOIN user_modules um ON u.id = um.user_id LEFT JOIN modules m ON um.module_id = m.id WHERE u.id = $1 GROUP BY u.id",
112    )
113    .bind(new_user.id)
114    .fetch_one(&mut *tx)
115    .await?;
116
117    tx.commit().await?;
118
119    Ok(Json(final_user))
120}

123pub async fn get_user(
124    State(state): State<AppState>,
125    caller: crate::webserver::auth::JwtUser,
126    axum::extract::Path(id): axum::extract::Path<String>,
127) -> Result<Json<User>, AppError> {
128    if caller.sub != id
129        && caller.user_id.map(|uid| uid.to_string()) != Some(id.clone())
130        && caller.role != "admin"
131        && caller.role != "support"
132    {
133        return Err(AppError::Forbidden("Access denied".to_string()));
134    }
135
136    let user = if let Ok(numeric_id) = id.parse::<i64>() {
137        sqlx::query_as::<_, User>(
138            "SELECT u.id, u.name, u.email, u.role, u.phone, u.description, u.is_suspended, u.client_log_level, u.keycloak_sub, ARRAY_AGG(m.name) FILTER (WHERE m.name IS NOT NULL) AS modules FROM users u LEFT JOIN user_modules um ON u.id = um.user_id LEFT JOIN modules m ON um.module_id = m.id WHERE u.id = $1 OR u.keycloak_sub = $2 GROUP BY u.id",
139        )
140        .bind(numeric_id)
141        .bind(&id)
142        .fetch_optional(&state.db_pool)
143        .await?
144    } else {
145        sqlx::query_as::<_, User>(
146            "SELECT u.id, u.name, u.email, u.role, u.phone, u.description, u.is_suspended, u.client_log_level, u.keycloak_sub, ARRAY_AGG(m.name) FILTER (WHERE m.name IS NOT NULL) AS modules FROM users u LEFT JOIN user_modules um ON u.id = um.user_id LEFT JOIN modules m ON um.module_id = m.id WHERE u.keycloak_sub = $1 GROUP BY u.id",
147        )
148        .bind(&id)
149        .fetch_optional(&state.db_pool)
150        .await?
151    };
152
153    match user {
154        Some(u) => Ok(Json(u)),
155        None => Err(AppError::NotFound(format!("User {id} not found"))),
156    }
157}

160pub async fn update_user(
161    State(state): State<AppState>,
162    caller: crate::webserver::auth::JwtUser,
163    raw_token: Option<axum::Extension<crate::webserver::auth::RawToken>>,
164    axum::extract::Path(id): axum::extract::Path<String>,
165    Json(user): Json<User>,
166) -> Result<Json<User>, AppError> {
167    if caller.sub != id
168        && caller.user_id.map(|uid| uid.to_string()) != Some(id.clone())
169        && caller.role != "admin"
170    {
171        return Err(AppError::Forbidden("Access denied".to_string()));
172    }
173
174    let mut tx = state.db_pool.begin().await?;
175
176    let existing_user = if let Ok(numeric_id) = id.parse::<i64>() {
177        sqlx::query_as::<_, User>(
178            "SELECT u.id, u.name, u.email, u.role, u.phone, u.description, u.is_suspended, u.client_log_level, u.keycloak_sub, ARRAY_AGG(m.name) FILTER (WHERE m.name IS NOT NULL) AS modules FROM users u LEFT JOIN user_modules um ON u.id = um.user_id LEFT JOIN modules m ON um.module_id = m.id WHERE u.id = $1 OR u.keycloak_sub = $2 GROUP BY u.id",
179        )
180        .bind(numeric_id)
181        .bind(&id)
182        .fetch_optional(&mut *tx)
183        .await?
184    } else {
185        sqlx::query_as::<_, User>(
186            "SELECT u.id, u.name, u.email, u.role, u.phone, u.description, u.is_suspended, u.client_log_level, u.keycloak_sub, ARRAY_AGG(m.name) FILTER (WHERE m.name IS NOT NULL) AS modules FROM users u LEFT JOIN user_modules um ON u.id = um.user_id LEFT JOIN modules m ON um.module_id = m.id WHERE u.keycloak_sub = $1 GROUP BY u.id",
187        )
188        .bind(&id)
189        .fetch_optional(&mut *tx)
190        .await?
191    }
192    .ok_or_else(|| AppError::NotFound(format!("User {id} not found")))?;
193
194    let db_id = existing_user.id;
195    let is_admin = caller.role == "admin";
196
197    let role_to_save = if is_admin {
198        &user.role
199    } else {
200        &existing_user.role
201    };
202    let is_suspended_to_save = if is_admin {
203        user.is_suspended
204    } else {
205        existing_user.is_suspended
206    };
207    let log_level_to_save = if is_admin {
208        if user.client_log_level.is_empty() {
209            "INFO"
210        } else {
211            &user.client_log_level
212        }
213    } else {
214        &existing_user.client_log_level
215    };
216    let modules_to_save = if is_admin {
217        &user.modules
218    } else {
219        &existing_user.modules
220    };
221    let keycloak_sub_to_save = user
222        .keycloak_sub
223        .as_ref()
224        .or(existing_user.keycloak_sub.as_ref());
225
226    sqlx::query(
227        "UPDATE users SET name = $1, email = $2, role = $3, phone = $4, description = $5, is_suspended = $6, client_log_level = $7, keycloak_sub = $8 WHERE id = $9",
228    )
229    .bind(&user.name)
230    .bind(&user.email)
231    .bind(role_to_save)
232    .bind(&user.phone)
233    .bind(&user.description)
234    .bind(is_suspended_to_save)
235    .bind(log_level_to_save)
236    .bind(keycloak_sub_to_save)
237    .bind(db_id)
238    .execute(&mut *tx)
239    .await?;
240
241    if is_admin {
242        if let Some(modules) = modules_to_save {
243            sqlx::query("DELETE FROM user_modules WHERE user_id = $1")
244                .bind(db_id)
245                .execute(&mut *tx)
246                .await?;
247            if !modules.is_empty() {
248                sqlx::query("INSERT INTO user_modules (user_id, module_id) SELECT $1, id FROM modules WHERE name = ANY($2)")
249                    .bind(db_id)
250                    .bind(modules)
251                    .execute(&mut *tx)
252                    .await?;
253            }
254        }
255    }
256
257    let raw_token_str = raw_token.map(|axum::Extension(rt)| rt.0);
258    let empty_modules = Vec::new();
259    let modules_slice = modules_to_save.as_ref().unwrap_or(&empty_modules);
260
261    crate::service::keycloak::sync_user_to_keycloak(
262        &state.config,
263        raw_token_str.as_deref(),
264        &user.email,
265        !is_suspended_to_save,
266        is_suspended_to_save,
267        modules_slice,
268    )
269    .await?;
270
271    tx.commit().await?;
272
273    let updated_user = sqlx::query_as::<_, User>(
274        "SELECT u.id, u.name, u.email, u.role, u.phone, u.description, u.is_suspended, u.client_log_level, u.keycloak_sub, ARRAY_AGG(m.name) FILTER (WHERE m.name IS NOT NULL) AS modules FROM users u LEFT JOIN user_modules um ON u.id = um.user_id LEFT JOIN modules m ON um.module_id = m.id WHERE u.id = $1 GROUP BY u.id",
275    )
276    .bind(db_id)
277    .fetch_one(&state.db_pool)
278    .await?;
279
280    Ok(Json(updated_user))
281}

284pub async fn delete_user(
285    State(state): State<AppState>,
286    caller: crate::webserver::auth::JwtUser,
287    axum::extract::Path(id): axum::extract::Path<String>,
288) -> Result<axum::http::StatusCode, AppError> {
289    if caller.sub != id
290        && caller.user_id.map(|uid| uid.to_string()) != Some(id.clone())
291        && caller.role != "admin"
292    {
293        return Err(AppError::Forbidden("Access denied".to_string()));
294    }
295
296    let env = state.config.debugging.environment.as_str();
297    let dev_auth = state.config.debugging.enable_dev_auth;
298    if env != "development" && env != "testing" && !dev_auth {
299        return Err(AppError::Forbidden(
300            "User deletion is disabled in this environment".to_string(),
301        ));
302    }
303
304    let result = if let Ok(numeric_id) = id.parse::<i64>() {
305        sqlx::query("DELETE FROM users WHERE id = $1 OR keycloak_sub = $2")
306            .bind(numeric_id)
307            .bind(&id)
308            .execute(&state.db_pool)
309            .await?
310    } else {
311        sqlx::query("DELETE FROM users WHERE keycloak_sub = $1")
312            .bind(&id)
313            .execute(&state.db_pool)
314            .await?
315    };
316
317    if result.rows_affected() == 0 {
318        return Err(AppError::NotFound("User not found".to_string()));
319    }
320
321    if let Ok(numeric_id) = id.parse::<i64>() {
322        state.farms_cache.write().await.remove(&numeric_id);
323    }
324    Ok(axum::http::StatusCode::NO_CONTENT)
